\section{Problema}
% ============================================================

% ---------- 3 · EL PROBLEMA ----------
\begin{frame}{El problema: se memoriza, no se concreta}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{itemize}
        \item En macro de pregrado se \textbf{derivan} modelos en la pizarra\ldots
              y ahí se quedan.
        \item Concretarlos exige \textbf{Dynare}, que casi nadie domina.
        \item El alumno no llega a \emph{experimentar} con el modelo que estudió.
      \end{itemize}
      \vspace{6pt}
      \begin{block}{Dolor validado (encuesta, n=59)}
        El \textbf{71\%} de los estudiantes de macro \textbf{nunca usó Dynare}.
      \end{block}
    \column{0.48\textwidth}
      \begin{center}
        \includegraphics[width=\columnwidth]{seg_nivel.png}\\
        {\scriptsize\color{UWGrayL3} Nivel de experiencia con Dynare (n=59).}
      \end{center}
  \end{columns}
\end{frame}

% ---------- 4 · WHY NOW / INSIGHT ----------
\begin{frame}{El insight: la IA genérica se equivoca y el alumno no lo nota}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{itemize}
        \item La IA general produce Dynare \textbf{plausible pero incorrecto}.
        \item El alumno no tiene cómo \textbf{verificar} si está bien.
        \item \textbf{Why now:} LLMs baratos (DeepSeek) + OCR gratis (PaddleOCR)
              hacen viable un laboratorio asistido y \emph{verificado}.
      \end{itemize}
      \vspace{6pt}
      \begin{alertblock}{La brecha}
        El \textbf{64\%} \textbf{no pudo verificar} si el código de la IA era correcto.
      \end{alertblock}
    \column{0.48\textwidth}
      \begin{center}
        \includegraphics[width=\columnwidth]{ia_corregir.png}\\
        {\scriptsize\color{UWGrayL3} ¿Pudiste verificar el código de la IA? (n=59).}
      \end{center}
  \end{columns}
\end{frame}

\section{Solución}
% ============================================================

% ---------- 5 · LA SOLUCIÓN ----------
\begin{frame}{La solución: MacroBuilt, tu laboratorio macroeconómico}
  Traduce las ecuaciones que el alumno deriva a mano y las convierte en
  \textbf{código Dynare correcto, verificado y explicado}.
  \vspace{8pt}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{Traduce}
        Ecuaciones $\rightarrow$ archivo \texttt{.mod} listo para correr.
      \end{block}
      \begin{exampleblock}{Verifica (sin LLM)}
        Chequeo \textbf{determinista}: conteo, símbolos, \emph{timing}, Blanchard--Kahn.
      \end{exampleblock}
    \column{0.5\textwidth}
      \begin{block}{Explica}
        Comenta cada bloque y ecuación con la intuición económica.
      \end{block}
      \begin{block}{Foto de pizarra + Bootcamp}
        OCR del manuscrito y un curso interactivo \textbf{autocorregido}.
      \end{block}
  \end{columns}
\end{frame}

% ---------- 6 · PRODUCTO Y ARQUITECTURA ----------
\begin{frame}{Producto y arquitectura: el moat es el verificador}
  \begin{center}
    \includegraphics[width=0.96\textwidth]{arquitectura.png}
  \end{center}
  \vspace{2pt}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{itemize}\small
        \item \textbf{No es un wrapper de un LLM:} el verificador es propio,

      \end{itemize}

        \item Validado contra \textbf{185 modelos publicados del MMB} +

  \end{columns}
\end{frame}
